const rangeTextDefaults = {"version":1,"eyebrow":"A WARDROBE FOR EVERY YOU","heading":"Explore our","accent":"range.","subtitle":"Your next favourite is waiting to be found.","footer":"A little preview of what’s inside.","buttonLabel":"Explore the collections","cards":{"range-everyday":{"title":"Casual Wear"},"range-work-edit":{"title":"Work Wear"},"range-occasion":{"title":"Occasional Wear"},"range-new-arrivals":{"title":"Ananya’s Pick"},"range-shop-all":{"title":"Ananya’s Pick"}}};
const rangeTextKeys = ["eyebrow", "heading", "accent", "subtitle", "footer", "buttonLabel"];
const rangeTextCardIds = ["range-everyday", "range-work-edit", "range-occasion", "range-new-arrivals", "range-shop-all"];
let rangeTextCurrent=structuredClone(rangeTextDefaults),rangeTextPrevious=null,rangeTextLoaded=false,rangeTextDirty=false,rangeTextBusy=false;
function rangeTextMessage(message,error=false){$("rangeTextStatus").textContent=message;$("rangeTextStatus").style.color=error ? "#9b2424" : "#602124";}
function rangeTextFormValue(){const value={cards:{}};for(const key of rangeTextKeys)value[key]=$("rangeText-"+key).value;for(const id of rangeTextCardIds)value.cards[id]={title:$("rangeText-"+id).value};return value;}
function rangeTextPreview(){
  const value=rangeTextFormValue(),host=$("rangeTextPreview");host.replaceChildren();
  const add=(tag,text)=>{const node=document.createElement(tag);node.textContent=text;host.append(node);return node;};
  add("p",value.eyebrow);const heading=add("h2",value.heading+" "),accent=document.createElement("em");accent.textContent=value.accent;heading.append(accent);add("p",value.subtitle);
  const captions=add("div","");captions.className="range-preview-captions";
  for(const id of rangeTextCardIds){const caption=document.createElement("span");caption.textContent=value.cards[id].title;captions.append(caption);}
  add("p",value.footer);add("p",value.buttonLabel+" →");
}
function rangeTextButtons(){
  $("rangeTextFields").disabled=!rangeTextLoaded || rangeTextBusy;
  $("rangeTextSave").disabled=demoMode || !rangeTextDirty || rangeTextBusy;
  $("rangeTextReset").disabled=demoMode || rangeTextBusy;
  $("rangeTextRestore").disabled=demoMode || !rangeTextPrevious || rangeTextBusy;
}
function rangeTextFill(value){
  rangeTextCurrent=value || structuredClone(rangeTextDefaults);
  for(const key of rangeTextKeys)$("rangeText-"+key).value=rangeTextCurrent[key] ?? rangeTextDefaults[key];
  for(const id of rangeTextCardIds)$("rangeText-"+id).value=rangeTextCurrent.cards?.[id]?.title || rangeTextDefaults.cards[id].title;
  rangeTextDirty=false;rangeTextLoaded=true;rangeTextPreview();rangeTextButtons();
}
async function loadRangeText(){
  if(demoMode){rangeTextFill(rangeTextDefaults);rangeTextMessage("Local text preview only.");return;}
  const response=await fetch("/api/hidi/landing-media-range-text",{cache:"no-store"});const data=await response.json().catch(()=>({}));
  if(!response.ok)throw new Error(data.message || "Unable to load range text");
  rangeTextPrevious=data.previous;rangeTextFill(data.current);rangeTextMessage("Current text loaded. Edit and Save Text to publish.");
}
async function changeRangeText(action){
  if(rangeTextBusy || !rangeTextLoaded || demoMode)return;
  if(action==="publish" && !$("rangeTextForm").reportValidity())return;
  if(action!=="publish" && rangeTextDirty && !window.confirm("Replace your unsaved text edits?"))return;
  rangeTextBusy=true;rangeTextButtons();rangeTextMessage("Saving text...");
  try{
    const before=rangeTextCurrent;
    const data=await request("/api/hidi/landing-media-range-text-"+action,action==="publish"?rangeTextFormValue():{});
    rangeTextPrevious=before;rangeTextFill(data.current);rangeTextMessage(action==="publish"?"Text saved and published. Refresh the storefront to see it.":action==="restore"?"Previous text restored and published.":"Default text restored and published.");
  }catch(error){rangeTextMessage(error.message || "Unable to save text",true);}
  finally{rangeTextBusy=false;rangeTextButtons();}
}
$("rangeTextForm").addEventListener("input",()=>{rangeTextDirty=true;rangeTextPreview();rangeTextButtons();rangeTextMessage("Unsaved text changes — use Save Text to publish.");});
$("rangeTextForm").addEventListener("submit",event=>{event.preventDefault();changeRangeText("publish");});
$("rangeTextReset").onclick=()=>changeRangeText("reset");$("rangeTextRestore").onclick=()=>changeRangeText("restore");
